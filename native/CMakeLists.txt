cmake_minimum_required(VERSION 3.24)
project(ZUINative LANGUAGES C)
get_filename_component(ZUI_LVGL_DIR "${ZUI_LVGL_DIR}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
if(NOT EXISTS "${ZUI_LVGL_DIR}/lvgl.h")
  message(FATAL_ERROR "LVGL source not found. Supply zig build -Dlvgl-dir=/path/to/lvgl-9.6.0")
endif()
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_INSTALL OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
find_package(SDL2 REQUIRED)
add_subdirectory("${ZUI_LVGL_DIR}" lvgl)
target_link_libraries(lvgl PUBLIC SDL2::SDL2)
add_library(zui_native STATIC host.c)
target_link_libraries(zui_native PUBLIC lvgl SDL2::SDL2)
target_include_directories(zui_native PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(zui_native PRIVATE -Wall -Wextra -Werror)
